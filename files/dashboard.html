// 특정 학생 학번(studentNum) 데이터만 삭제하는 함수 window.deleteStudentData = function(studentNum) { if (!confirm(`${studentNum} 학생의 제출 기록을 삭제하시겠습니까?`)) return; const selectedUnit = document.getElementById('unitSelect').value; if (selectedUnit === '2단원') { // Firebase에서 해당 학번을 가진 데이터의 Key를 찾아 삭제 const dbRef = ref(db, 'yeep_activity_submissions'); onValue(dbRef, (snapshot) => { const val = snapshot.val(); if (val) { Object.keys(val).forEach(key => { const item = val[key]; const rawNum = item.studentInfo || item.num || item.studentNum || ''; if (String(rawNum) === String(studentNum)) { remove(ref(db, `yeep_activity_submissions/${key}`)) .then(() => alert(`${studentNum} 학생 데이터가 삭제되었습니다.`)) .catch((err) => alert("삭제 실패: " + err.message)); } }); } }, { onlyOnce: true }); } else { // 1, 3단원 (로컬스토리지) 개별 삭제 let localData = getLocalSubmissions(); localData = localData.filter(item => String(item.num) !== String(studentNum)); localStorage.setItem('career_submissions', JSON.stringify(localData)); alert(`${studentNum} 학생 데이터가 삭제되었습니다.`); renderDashboard(); } };
